def apply():
    from mlx_lm.models.muse_glimmer import Attention, MuseGlimmerModel
    import mlx.core as mx

    orig_attn_call = Attention.__call__
    def fused_attn_call(self, x, mask=None, cache=None):
        # if fused weight not present, fallback
        if not hasattr(self, '_qkv_fused_weight'):
            return orig_attn_call(self, x, mask, cache)
        B, L, _ = x.shape
        # fused projection
        proj = x @ self._qkv_fused_weight.T
        q_out = self.n_heads * self.head_dim
        k_out = self.n_kv_heads * self.head_dim
        # v_out same as k_out
        q = proj[..., :q_out].reshape(B, L, self.n_heads, self.head_dim)
        k = proj[..., q_out:q_out+k_out].reshape(B, L, self.n_kv_heads, self.head_dim)
        v = proj[..., q_out+k_out:q_out+2*k_out].reshape(B, L, self.n_kv_heads, self.head_dim)

        # Scaleless QK-norm
        q = mx.fast.rms_norm(q, None, self.eps) * self.qk_scale_factor
        k = mx.fast.rms_norm(k, None, self.eps)

        q = q.transpose(0,2,1,3)
        k = k.transpose(0,2,1,3)
        v = v.transpose(0,2,1,3)

        if self.use_rope:
            offset = cache.offset if cache is not None else 0
            q = self.rope(q, offset=offset)
            k = self.rope(k, offset=offset)

        if cache is not None:
            k, v = cache.update_and_fetch(k, v)

        from mlx_lm.models.base import scaled_dot_product_attention
        out = scaled_dot_product_attention(q, k, v, cache=cache, scale=self.scale, mask=mask)
        out = out.transpose(0,2,1,3).reshape(B, L, -1)

        out = out * mx.sigmoid(self.gate_proj(x))
        return self.o_proj(out)

    Attention.__call__ = fused_attn_call

    # Patch MuseGlimmerModel to skip embedding rms_norm if pre-normalized
    orig_muse_call = MuseGlimmerModel.__call__
    def muse_call(self, inputs, cache=None):
        # embedding
        h = self.embed_tokens(inputs)
        # if pre-normalized, skip rms_norm
        if getattr(self, '_embed_normalized', False):
            # h already normalized
            pass
        else:
            h = mx.fast.rms_norm(h, None, self.embed_eps)
        if cache is None:
            cache = [None] * len(self.layers)
        masks = self._masks(h, cache)
        for layer, c, mask in zip(self.layers, cache, masks):
            h = layer(h, mask, c)
        return self.norm(h)
    MuseGlimmerModel.__call__ = muse_call
