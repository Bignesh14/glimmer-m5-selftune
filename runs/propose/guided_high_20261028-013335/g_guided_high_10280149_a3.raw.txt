def apply():
    import mlx_lm.models.muse_glimmer as mg
    from mlx_lm.models.muse_glimmer import Attention, MLP
    import mlx.core as mx

    # Patch Attention.__call__
    original_attention_call = Attention.__call__

    def fused_attention_call(self, x, mask=None, cache=None):
        B, L, _ = x.shape
        # Use fused weight if prepared
        if hasattr(self, '_fused_qkv_gate_weight'):
            # single matmul
            # x @ W.T
            combined = mx.matmul(x, self._fused_qkv_gate_weight.T)
            q_size = self.n_heads * self.head_dim
            kv_size = self.n_kv_heads * self.head_dim
            # split
            q = combined[..., :q_size]
            k = combined[..., q_size:q_size+kv_size]
            v = combined[..., q_size+kv_size:q_size+2*kv_size]
            gate_input = combined[..., q_size+2*kv_size:q_size+2*kv_size+q_size]
        else:
            # fallback
            q = self.q_proj(x)
            k = self.k_proj(x)
            v = self.v_proj(x)
            gate_input = self.gate_proj(x)

        # reshape
        q = q.reshape(B, L, self.n_heads, self.head_dim)
        k = k.reshape(B, L, self.n_kv_heads, self.head_dim)
        v = v.reshape(B, L, self.n_kv_heads, self.head_dim)

        # rms norm
        q = mx.fast.rms_norm(q, None, self.eps) * self.qk_scale_factor
        k = mx.fast.rms_norm(k, None, self.eps)

        q = q.transpose(0,2,1,3)
        k = k.transpose(0,2,1,3)
        v = v.transpose(0,2,1,3)

        if self.use_rope:
            offset = cache.offset if cache is not None else 0
            q = self.rope(q, offset=offset)
            k = self.rope(k, offset=offset)

        if cache is not None:
            k, v = cache.update_and_fetch(k, v)

        out = mg.scaled_dot_product_attention(q, k, v, cache=cache, scale=self.scale, mask=mask)
        out = out.transpose(0,2,1,3).reshape(B, L, -1)

        # gate
        out = out * mx.sigmoid(gate_input)
        return self.o_proj(out)

    Attention.__call__ = fused_attention_call

    # Patch MLP.__call__
    original_mlp_call = MLP.__call__

    def fused_mlp_call(self, x):

            up = combined[..., mid:]
            # silu
            return self.down_proj(mx.silu(gate) * up)
        else:
            return original_mlp_call(self, x)

    MLP.__call__ = fused_mlp_call
