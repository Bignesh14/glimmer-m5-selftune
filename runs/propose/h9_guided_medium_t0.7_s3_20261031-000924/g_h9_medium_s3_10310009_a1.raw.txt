def __call__(self, x: mx.array, mask=None, cache=None) -> mx.array:
    B, L, _ = x.shape

    q = self.q_proj(x).reshape(B, L, self.n_heads, self.head_dim)
    k = self.k_proj(x).reshape(B, L, self.n_kv_heads, self.head_dim)
    v = self.v_proj(x).reshape(B, L, self.n_kv_heads, self.head_dim)

    # Scaleless QK-norm over head_dim; Q additionally scaled.
    q = mx.fast.rms_norm(q, None, self.eps) * self.qk_scale_factor
    k = mx.fast.rms_norm(k, None, self.eps)

    q = q.transpose(0, 2, 1, 3)
    k = k.transpose(0, 2, 1, 3)
    v = v.transpose(0, 2, 1, 3)

    if self.use_rope:
        offset = cache.offset if cache is not None else 0
        q = self.rope(q, offset=offset)
        k = self.rope(k, offset=offset)

    if cache is not None:
        k, v = cache.update_and_fetch(k, v)

    out = scaled_dot_product_attention(
        q, k, v, cache=cache, scale=self.scale, mask=mask
    )
    out = out.transpose(0, 2, 1, 3).reshape(B, L, -1)

    # Glimmer's output gate.
    out = out * mx.sigmoid(self.gate_proj(x))
    return self.o_proj(out)

"""head_split_fused_sdpa — split query heads to stay inside the fused SDPA kernel"""

NAME = "head_split_fused_sdpa"
EXACT = True
DESCRIPTION = "Split query heads for 3-8 token passes so query_length * heads/2 <=32, keeping fused SDPA. One-token and prompt processing unchanged."

def apply():
    import mlx_lm.models.muse_glimmer as mg
    from mlx_lm.models.base import scaled_dot_product_attention
    import mlx.core as mx

    original_call = mg.Attention.__call__

    def new_call(self, x: mx.array, mask=None, cache=None) -> mx.array:
        B, L, _ = x.shape

        q = self.q_proj(x).reshape(B, L, self.n_heads, self.head_dim)
        k = self.k_proj(x).reshape(B, L, self.n_kv_heads, self.head_dim)
        v = self.v_proj(x).reshape(B, L, self.n_kv_heads, self.head_dim)

        q = mx.fast.rms_norm(q, None, self.eps) * self.qk_scale_factor
        k = mx.fast.rms_norm(k, None, self.eps)

        q = q.transpose(0, 2, 1, 3)
        k = k.transpose(0, 2, 1, 3)
        v = v.transpose(0, 2, 1, 3)

        if self.use_rope:
            offset = cache.offset if cache is not None else 0
            q = self.rope(q, offset=offset)
            k = self.rope(k, offset=offset)

        if cache is not None:
            k, v = cache.update_and_fetch(k, v)

        # Use head splitting only for multi-token passes 3..8 to hit the fused kernel
        # Condition for fused kernel: query_length * (heads_per_call / kv_heads) <= 32
        # With kv_heads=2 -> heads_per_call <= 64 / L
        if 3 <= L <= 8:
            n_heads = self.n_heads
            max_heads_per_call = 64 // L
            if max_heads_per_call < n_heads and max_heads_per_call > 0:
                chunk = max_heads_per_call
                outs = []
                for start in range(0, n_heads, chunk):
                    end = start + chunk
                    if end > n_heads:
                        end = n_heads
                    q_chunk = q[:, start:end, :, :]
                    out_chunk = scaled_dot_product_attention(
                        q_chunk, k, v, cache=cache, scale=self.scale, mask=mask
                    )
                    outs.append(out_chunk)
                out = mx.concatenate(outs, axis=1)
            else:
                out = scaled_dot_product_attention(
                    q, k, v, cache=cache, scale=self.scale, mask=mask
                )
        else:
            out = scaled_dot_product_attention(
                q, k, v, cache=cache, scale=self.scale, mask=mask
            )

        out = out.transpose(0, 2, 1, 3).reshape(B, L, -1)
        out = out * mx.sigmoid(self.gate_proj(x))
        return self.o_proj(out)

    mg.Attention.__call__ = new_call
